function buildPaginationItems(
  currentPage: number,
  totalPages: number,
): PaginationItem[] {
  if (totalPages <= 7) {
    return Array.from(
      { length: totalPages },
      (_, index) => index + 1,
    );
  }

  const items: PaginationItem[] = [1];
  const start = Math.max(2, currentPage - 1);
  const end = Math.min(totalPages - 1, currentPage + 1);

  if (start > 2) {
    items.push("ellipsis");
  }

  for (let page = start; page <= end; page += 1) {
    items.push(page);
  }

  if (end < totalPages - 1) {
    items.push("ellipsis");
  }

  items.push(totalPages);

  return items;
}

function resolveImageUrl(value: string | null | undefined) {
  if (!value) {
    return "";
  }

  let url = value.trim().replace(/\\/g, "/");

  if (
    /^https?:\/\//i.test(url) ||
    url.startsWith("data:") ||
    url.startsWith("blob:")
  ) {
    return url;
  }

  url = url
    .replace(/^\.\//, "")
    .replace(/^public\//i, "")
    .replace(/^\/public\//i, "/");

  const publicIndex = url.toLowerCase().indexOf("/public/");

  if (publicIndex >= 0) {
    url = url.slice(publicIndex + 8);
  }

  return url.startsWith("/") ? url : `/${url}`;
}

function imageValidation(files: File[]) {
  const allowed = [
    "image/jpeg",
    "image/png",
    "image/webp",

  const invalid = files.find(

  }

  return "";
}

const IMAGE_TARGET_BYTES = 1_900_000;
const IMAGE_MAX_SOURCE_BYTES = 25 * 1024 * 1024;
const IMAGE_MAX_DIMENSION = 2400;

function canvasToBlob(
  canvas: HTMLCanvasElement,
  type: string,
  quality: number,
): Promise<Blob> {
  return new Promise((resolve, reject) => {
    canvas.toBlob(
      (blob) => {
        if (!blob) {
          reject(
            new Error("Failed to process image."),
          );
          return;
        }

        resolve(blob);
      },
      type,
      quality,
    );
  });
}

async function compressImageTo2MB(
  file: File,
): Promise<File> {
  if (file.size > IMAGE_MAX_SOURCE_BYTES) {
    throw new Error(
      `${file.name} is larger than 25 MB. Please choose a smaller image.`,
    );
  }

  const objectUrl = URL.createObjectURL(file);

  try {
    const image = new window.Image();

    const loaded = new Promise<void>(
      (resolve, reject) => {
        image.onload = () => resolve();

        image.onerror = () =>
          reject(
            new Error(
              `Could not read ${file.name}.`,
            ),
          );
      },
    );

    image.src = objectUrl;

    await loaded;

    let width = image.naturalWidth;
    let height = image.naturalHeight;

    const longestSide = Math.max(
      width,
      height,
    );

    if (
      longestSide >
      IMAGE_MAX_DIMENSION
    ) {
      const scale =
        IMAGE_MAX_DIMENSION /
        longestSide;

      width = Math.max(
        1,
        Math.round(width * scale),
      );

      height = Math.max(
        1,
        Math.round(height * scale),
      );
    }

    const canvas =
      document.createElement("canvas");

    let quality = 0.82;

    for (
      let attempt = 0;
      attempt < 12;
      attempt += 1
    ) {
      canvas.width = width;
      canvas.height = height;

      const context =
        canvas.getContext("2d");

      if (!context) {
        throw new Error(
          "Your browser could not prepare the image.",
        );
      }

      context.clearRect(
        0,
        0,
        width,
        height,
      );

      context.drawImage(
        image,
        0,
        0,
        width,
        height,
      );

      const blob =
        await canvasToBlob(
          canvas,
          "image/webp",
          quality,
        );

      if (
        blob.size <=
        IMAGE_TARGET_BYTES
      ) {
        const baseName =
          file.name.replace(
            /\.[^/.]+$/,
            "",
          );

        return new File(
          [blob],
          `${baseName}.webp`,
          {
            type: "image/webp",
            lastModified:
              Date.now(),
          },
        );
      }

      if (quality > 0.5) {
        quality = Math.max(
          0.5,
          quality - 0.07,
        );
      } else {
        width = Math.max(
          640,
          Math.round(width * 0.85),
        );

        height = Math.max(
          640,
          Math.round(height * 0.85),
        );

        quality = 0.78;
      }
    }

    throw new Error(
      `${file.name} could not be compressed below 2 MB.`,
    );
  } finally {
    URL.revokeObjectURL(objectUrl);
  }
}

  async function handleImageSelection(
  files: FileList | null,
) {
  if (!files) {
    return;
  }

  const chosen = Array.from(files);

  if (
    selectedImages.length +
      chosen.length >
    5
  ) {
    setError(
      "You can select a maximum of 5 images.",
    );
    return;
  }

  const validationError =
    imageValidation(chosen);

  if (validationError) {
    setError(validationError);
    return;
  }

  setCompressingImages(true);
  setError("");
  setSuccess("");

  try {
    const compressedImages: SelectedImage[] =
      [];

    for (const file of chosen) {
      const compressedFile =
        await compressImageTo2MB(file);

      compressedImages.push({
        file: compressedFile,
        preview:
          URL.createObjectURL(
            compressedFile,
          ),
      });
    }

    setSelectedImages((current) => [
      ...current,
      ...compressedImages,
    ]);
  } catch (err) {
    setError(
      err instanceof Error
        ? err.message
        : "Failed to optimize product images.",
    );
  } finally {
    setCompressingImages(false);
  }
}

          `/api/admin/products/${productId}/images`,
          {
            method: "POST",
            body: formData,
          },
        );

      if (response.status === 401) {
        router.push("/admin/login");
        return;
      }

      const data =
        await response.json();

      if (!response.ok) {
        throw new Error(
          data.error ||
            data.message ||
            "Failed to upload product images.",
        );
      }
    } finally {
      setUploadingProductId(
        null,
      );
    }
  }

    async function handleImageUpload(
    productId: number,
    files: FileList | null,
  ) {
    if (!files?.length) {
      return;
    }

    const chosen =
      Array.from(files);

    if (chosen.length > 5) {
      setError(
        "You can upload a maximum of 5 images at a time.",
      );

      return;
    }

    const validationError =
      imageValidation(chosen);

    if (validationError) {
      setError(validationError);
      return;
    }

    setError("");

    try {
      await uploadImages(
        productId,
        chosen,
      );

      setSuccess(
        "Product images uploaded successfully.",
      );

      await refreshProducts();
    } catch (err) {
      setError(
        err instanceof Error
          ? err.message
          : "Failed to upload images.",
      );
    }
  }

  async function handleSubmit(
    event: FormEvent<HTMLFormElement>,
  ) {
    event.preventDefault();

    if (saving) {
      return;
    }

    setSaving(true);

    setError("");

    setSuccess("");

    try {
      const name =
        form.name.trim();

      if (!name) {
        throw new Error(
          "Product name is required.",
        );
      }

      const price =
        Number(form.price);

      if (
        !form.price.trim() ||
        !Number.isFinite(price) ||
        price < 0
      ) {
        throw new Error(
          "Please enter a valid selling price.",
        );
      }

      if (!selectedMainCategory) {
        throw new Error(
          "Please select a main category.",
        );
      }

      const categoryId =
        selectedSubcategory
          ? Number(
              selectedSubcategory,
            )
          : Number(
              selectedMainCategory,
            );

      if (
        !Number.isInteger(
          categoryId,
        ) ||
        categoryId <= 0
      ) {
        throw new Error(
          "Please select a valid category.",
        );
      }

      const compareAtPrice =
        form.compareAtPrice.trim()
          ? Number(
              form.compareAtPrice,
            )
          : null;

      if (
        compareAtPrice !== null &&
        (!Number.isFinite(
          compareAtPrice,
        ) ||
          compareAtPrice < 0)
      ) {
        throw new Error(
          "Please enter a valid compare-at price.",
        );
      }

      const response =
        await fetch(
          "/api/admin/products/create",
          {
            method: "POST",
            headers: {
              "Content-Type":
                "application/json",
            },
            body: JSON.stringify({
              barcode:
                form.barcode.trim() ||
                null,
              brand:
                form.brand.trim() ||
                null,
              name,
              sku:
                form.sku.trim() ||
                null,
              price,
              compareAtPrice,
              shortDescription:
                form.shortDescription.trim() ||
                null,
              description:
                form.description.trim() ||
                null,
              status:
                form.status,
              isAvailable:
                form.isAvailable,
              isFeatured:
                form.isFeatured,
              categoryId,
            }),
          },
        );

      if (response.status === 401) {
        router.push(
          "/admin/login",
        );

        return;
      }

      const data =
        await response.json();

      if (
        !response.ok ||
        !data.success
      ) {
        throw new Error(
          data.message ||
            data.error ||
            "Failed to create product.",
        );
      }

      const productId =
        Number(
          data.product?.id ??
            data.id,
        );

      if (
        !Number.isInteger(
          productId,
        ) ||
        productId <= 0
      ) {
        throw new Error(
          "The product API did not return a valid product ID.",
        );
      }

      let imagesUploaded =
        selectedImages.length ===
        0;

      if (
        selectedImages.length
      ) {
        try {
          await uploadImages(
            productId,
            selectedImages.map(
              (image) =>
                image.file,
            ),
          );

          imagesUploaded = true;
        } catch (err) {
          setError(
            err instanceof Error
              ? err.message
              : "The product was created, but image upload failed.",
          );
        }
      }

      selectedImages.forEach(
        (image) =>
          URL.revokeObjectURL(
            image.preview,
          ),
      );

      setSelectedImages([]);

      setForm(EMPTY_FORM);

      setSelectedMainCategory("");

      setSelectedSubcategory("");

      setGlobalProductFound(
        null,
      );

      setModalOpen(false);

      if (imagesUploaded) {
        setSuccess(
          selectedImages.length
            ? "Product and images saved successfully."
            : "Product created successfully.",
        );
      }

      await refreshProducts();
    } catch (err) {
      setError(
        err instanceof Error
          ? err.message
          : "Failed to create product.",
      );
    } finally {
      setSaving(false);
    }
  }

              >
                Next
              </button>
            </div>
          </div>
        ) : null}
      </section>

      <div className="mt-4 px-1 text-[9px] text-black/24">
        Click a product name to open its details.
      </div>

      {/* ADD PRODUCT MODAL */}

      {modalOpen ? (
        <div
          className="fixed inset-0 z-50 flex items-center justify-center overflow-y-auto bg-black/35 p-3 backdrop-blur-[3px] sm:p-6"
          onMouseDown={(event) => {
            if (
              event.target ===
                event.currentTarget &&
              !saving
            ) {
              closeCreateModal();
            }
          }}
        >
          <div
            ref={modalRef}
            className="w-full max-w-[820px] overflow-hidden rounded-2xl border border-black/8 bg-white shadow-[0_32px_100px_rgba(0,0,0,0.24)]"
          >
            <div className="relative flex items-center justify-between overflow-hidden border-b border-white/10 bg-[#111111] px-5 py-4 text-white">
              <div className="absolute inset-x-0 top-0 h-[3px] bg-[#d71920]" />

              <div className="pt-0.5">
                <div className="flex items-center gap-2">
                  <span className="text-[9px] font-bold uppercase tracking-[0.18em] text-white">
                    Catalog
                  </span>

                  <span className="size-1 rounded-full bg-[#d71920]" />

                  <span className="text-[8px] font-medium uppercase tracking-[0.14em] text-white/45">
                    Product setup
                  </span>
                </div>

                <h2 className="mt-1 text-[19px] font-semibold tracking-[-0.025em] text-white">
                  Add product
                </h2>

                <p className="mt-0.5 text-[10px] text-white/55">
                  Search the global catalog or enter product details manually.
                </p>
              </div>

              <button
                type="button"
                onClick={closeCreateModal}
                disabled={saving}
                className="grid size-9 place-items-center rounded-xl border border-[#d71920] bg-[#d71920] text-white shadow-[0_6px_18px_rgba(215,25,32,0.25)] transition hover:bg-[#bd151c] hover:border-[#bd151c] disabled:cursor-not-allowed disabled:opacity-40"

                <X size={16} strokeWidth={2.2} />
              </button>
            </div>

            <form
              onSubmit={

            >

                <div className="mb-4 rounded-xl border border-red-200 bg-red-50 px-3.5 py-2.5 text-[11px] text-red-700">
                  {error}
                </div>
              ) : null}

              <div className="grid gap-4 md:grid-cols-2 rounded-2xl bg-[#fffdfd]">
                {/* BARCODE */}

                <div className="md:col-span-2 overflow-hidden rounded-2xl border border-[#d71920]/20 bg-white shadow-[0_10px_30px_rgba(215,25,32,0.07)]">
                  <div className="flex items-center justify-between border-b border-[#d71920]/10 bg-[#fff7f7] px-4 py-3">
                    <div className="flex items-center gap-2.5">
                      <span className="grid size-8 place-items-center rounded-lg bg-[#d71920]/10 text-[#d71920]">

                      </span>

                      <div>
                        <p className="text-[10px] font-bold uppercase tracking-[0.16em] text-[#b9141a]">
                          Global catalog lookup
                        </p>

                        <p className="mt-0.5 text-[9px] text-black/35">
                          Scan a barcode to automatically load product details
                        </p>
                      </div>
                    </div>

                    <span className="rounded-full border border-[#d71920]/15 bg-white px-2.5 py-1 text-[8px] font-bold uppercase tracking-[0.12em] text-[#d71920]">
                      Barcode
                    </span>
                  </div>

                  <div className="p-4">
                    <div className="flex flex-col gap-2.5 sm:flex-row">
                      <div className="relative flex-1">
                        <span
                          aria-hidden="true"
                          className="pointer-events-none absolute left-3.5 top-1/2 z-10 grid size-6 -translate-y-1/2 place-items-center rounded-md bg-[#fff1f2] text-[#d71920]"
                        >
                          <Search size={13} strokeWidth={2.3} />
                        </span>

                        <input
                          ref={barcodeRef}
                          value={form.barcode}
                          onChange={(event) => {
                            setGlobalProductFound(null);
                            setSuccess("");
                            updateForm(
                              "barcode",
                              event.target.value,
                            );
                          }}

                            if (event.key !== "Enter") {

                            }

                                result === "found" ||
                                result === "missing"
                              ) {
                                requestAnimationFrame(() =>
                                  skuRef.current?.focus(),
                                );
                              }
                            })();
                          }}
                          className="admin-product-input h-11 w-full pr-3"
                          style={{ paddingLeft: "3.1rem" }}
                          placeholder="Scan or enter barcode"
                          inputMode="numeric"
                          autoComplete="off"
                          aria-label="Barcode"
                        />
                      </div>

                      <button
                        type="button"
                        onClick={() =>
                          void lookupGlobalProduct()
                        }
                        disabled={
                          globalLookupLoading ||
                          !form.barcode.trim()
                        }
                        className="inline-flex h-11 shrink-0 items-center justify-center gap-2 rounded-xl bg-[#d71920] px-5 text-[11px] font-bold text-white shadow-[0_6px_16px_rgba(215,25,32,0.20)] transition hover:bg-[#bd151c] disabled:cursor-not-allowed disabled:opacity-40"
                      >
                        {globalLookupLoading ? (
                          <RefreshCw
                            size={13}
                            className="animate-spin"
                          />
                        ) : (
                          <Search size={13} />
                        )}

                        {globalLookupLoading
                          ? "Finding..."
                          : "Find product"}
                      </button>
                    </div>

                    {globalProductFound === true ? (
                      <div className="mt-3 flex items-center gap-2 rounded-lg border border-emerald-200 bg-emerald-50 px-3 py-2">
                        <CircleCheck
                          size={14}
                          className="shrink-0 text-emerald-600"
                        />

                        <p className="text-[9px] font-semibold text-emerald-700">
                          Global product found. Product details have been loaded automatically.
                        </p>
                      </div>
                    ) : globalProductFound === false ? (
                      <div className="mt-3 flex items-center gap-2 rounded-lg border border-amber-200 bg-amber-50 px-3 py-2">
                        <span className="grid size-5 shrink-0 place-items-center rounded-full bg-amber-100 text-[10px] font-bold text-amber-700">
                          !
                        </span>

                        <p className="text-[9px] font-medium text-amber-700">
                          No global product found. Continue entering the product manually.
                        </p>
                      </div>
                    ) : (
                      <p className="mt-2 px-1 text-[9px] text-black/30">
                        Scan or enter a barcode, then press Enter to search.
                      </p>
                    )}
                  </div>
                </div>

                {/* SKU */}

                <FormField label="SKU">
                  <input
                    ref={skuRef}
                    value={form.sku}
                    onChange={(event) =>
                      updateForm(
                        "sku",
                        event.target.value,
                      )
                    }
                    onKeyDown={(event) =>
                      focusNextOnEnter(
                        event,
                        nameRef,
                      )
                    }
                    className="admin-product-input"
                    placeholder="e.g. KTL-001"
                  />
                </FormField>

                {/* PRODUCT NAME */}

                <FormField label="Product name *">
                  <input
                    ref={nameRef}
                    value={form.name}
                    onChange={(event) =>
                      updateForm(
                        "name",
                        event.target.value,
                      )
                    }
                    onKeyDown={(event) =>
                      focusNextOnEnter(
                        event,
                        brandRef,
                      )
                    }
                    className="admin-product-input"
                    placeholder="e.g. Stainless Steel Kettle"
                    required
                  />
                </FormField>

                {/* BRAND */}

                <FormField label="Brand">
                  <input
                    ref={brandRef}
                    value={form.brand}
                    onChange={(event) =>
                      updateForm(
                        "brand",
                        event.target.value,
                      )
                    }
                    onKeyDown={(event) =>
                      focusNextOnEnter(
                        event,
                        mainCategoryRef,
                      )
                    }
                    className="admin-product-input"
                    placeholder="e.g. Philips"
                  />
                </FormField>

                {/* MAIN CATEGORY */}

                <FormField label="Main category *">
                  <select
                    ref={mainCategoryRef}
                    value={
                      selectedMainCategory
                    }
                    onChange={(event) => {
                      setSelectedMainCategory(
                        event.target.value,
                      );

                      setSelectedSubcategory(
                        "",
                      );
                    }}
                    onKeyDown={(event) => {
                      if (
                        event.key !==
                        "Enter"
                      ) {
                        return;
                      }

                      event.preventDefault();

                      subcategoryRef.current?.focus();
                    }}
                    disabled={
                      loadingCategories
                    }
                    required
                    className="admin-product-input"
                  >
                    <option value="">
                      {loadingCategories
                        ? "Loading categories..."
                        : "Select main category"}
                    </option>

                    {categories
                      .filter(
                        (category) =>
                          category.active !==
                          false,
                      )
                      .map(
                        (
                          category,
                        ) => (
                          <option
                            key={
                              category.id
                            }
                            value={String(
                              category.id,
                            )}
                          >
                            {
                              category.name
                            }
                          </option>
                        ),
                      )}
                  </select>
                </FormField>

                {/* SUBCATEGORY */}

                <FormField label="Subcategory">
                  <select
                    ref={subcategoryRef}
                    value={
                      selectedSubcategory
                    }
                    onChange={(event) =>
                      setSelectedSubcategory(
                        event.target.value,
                      )
                    }
                    onKeyDown={(event) => {
                      if (
                        event.key !==
                        "Enter"
                      ) {
                        return;
                      }

                      event.preventDefault();

                      priceRef.current?.focus();
                    }}
                    disabled={
                      !selectedMain
                    }
                    className="admin-product-input"
                  >
                    <option value="">
                      {!selectedMain
                        ? "Select main category first"
                        : (
                            selectedMain.children ??
                            []
                          ).length
                          ? "Use main category"
                          : "No subcategories"}
                    </option>

                    {(
                      selectedMain?.children ??
                      []
                    )
                      .filter(
                        (child) =>
                          child.active !==
                          false,
                      )
                      .map(
                        (
                          child,
                        ) => (
                          <option
                            key={
                              child.id
                            }
                            value={String(
                              child.id,
                            )}
                          >
                            {
                              child.name
                            }
                          </option>
                        ),
                      )}
                  </select>
                </FormField>

                {/* SELLING PRICE */}

                <FormField label="Selling price (LKR) *">
                  <input
                    ref={priceRef}
                    value={form.price}
                    onChange={(event) =>
                      updateForm(
                        "price",
                        event.target.value,
                      )
                    }
                    onKeyDown={(event) =>
                      focusNextOnEnter(
                        event,
                        compareAtPriceRef,
                      )
                    }
                    className="admin-product-input"
                    type="number"
                    min="0"
                    step="0.01"
                    inputMode="decimal"
                    placeholder="8700"
                    required
                  />
                </FormField>

                {/* COMPARE-AT PRICE */}

                <FormField label="Compare-at price (LKR)">
                  <input
                    ref={
                      compareAtPriceRef
                    }
                    value={
                      form.compareAtPrice
                    }
                    onChange={(event) =>
                      updateForm(
                        "compareAtPrice",
                        event.target.value,
                      )
                    }
                    onKeyDown={(event) =>
                      focusNextOnEnter(
                        event,
                        statusRef,
                      )
                    }
                    className="admin-product-input"
                    type="number"
                    min="0"
                    step="0.01"
                    inputMode="decimal"
                    placeholder="12500"
                  />
                </FormField>

                {/* STATUS */}

                <FormField label="Status">
                  <select
                    ref={statusRef}
                    value={form.status}
                    onChange={(event) =>
                      updateForm(
                        "status",
                        event.target.value as ProductStatus,
                      )
                    }
                    onKeyDown={(event) => {
                      if (
                        event.key !==
                        "Enter"
                      ) {
                        return;
                      }

                      event.preventDefault();

                      availableRef.current?.focus();
                    }}
                    className="admin-product-input"
                  >
                    <option value="DRAFT">
                      Draft
                    </option>

                    <option value="PUBLISHED">
                      Published
                    </option>

                    <option value="ARCHIVED">
                      Archived
                    </option>
                  </select>
                </FormField>

                {/* TOGGLES */}

                <div className="grid gap-3 md:grid-cols-2 md:col-span-2">
                  <ToggleField
                    label="Available"
                    type="available"
                    checked={
                      form.isAvailable
                    }
                    inputRef={
                      availableRef
                    }
                    onKeyDown={(event) => {
                      if (
                        event.key !==
                        "Enter"
                      ) {
                        return;
                      }

                      event.preventDefault();

                      updateForm(
                        "isAvailable",
                        !form.isAvailable,
                      );

                      requestAnimationFrame(
                        () =>
                          featuredRef.current?.focus(),
                      );
                    }}
                    onChange={(value) =>
                      updateForm(
                        "isAvailable",
                        value,
                      )
                    }
                  />

                  <ToggleField
                    label="Featured"
                    type="featured"
                    checked={
                      form.isFeatured
                    }
                    inputRef={
                      featuredRef
                    }
                    onKeyDown={(event) => {
                      if (
                        event.key !==
                        "Enter"
                      ) {
                        return;
                      }

                      event.preventDefault();

                      updateForm(
                        "isFeatured",
                        !form.isFeatured,
                      );

                      requestAnimationFrame(
                        () =>
                          shortDescriptionRef.current?.focus(),
                      );
                    }}
                    onChange={(value) =>
                      updateForm(
                        "isFeatured",
                        value,
                      )
                    }
                  />
                </div>

                {/* SHORT DESCRIPTION */}

                <div className="md:col-span-2">
                  <FormField label="Short description">
                    <textarea
                      ref={
                        shortDescriptionRef
                      }
                      value={
                        form.shortDescription
                      }
                      onChange={(event) =>
                        updateForm(
                          "shortDescription",
                          event.target.value,
                        )
                      }
                      className="admin-product-input min-h-20 resize-y py-2.5"
                      placeholder="Short product summary"
                    />
                  </FormField>
                </div>

                {/* DESCRIPTION */}

                <div className="md:col-span-2">
                  <FormField label="Description">
                    <textarea
                      value={
                        form.description
                      }
                      onChange={(event) =>
                        updateForm(
                          "description",
                          event.target.value,
                        )
                      }
                      className="admin-product-input min-h-28 resize-y py-2.5"
                      placeholder="Detailed product information"
                    />
                  </FormField>
                </div>

                {/* IMAGES */}

                <div className="rounded-xl border border-black/7 bg-[#fafafa] p-3.5 md:col-span-2">
  <div className="mb-3 flex items-center justify-between">
    <div>
      <p className="text-[9px] font-semibold uppercase tracking-[0.15em] text-black/35">
        Product images
      </p>

      <p className="mt-0.5 text-[10px] text-black/30">
        JPG, PNG or WebP â€¢ auto-compressed to under 2 MB â€¢ maximum 5
      </p>
    </div>

    <span className="rounded-full border border-black/6 bg-white px-2 py-1 text-[9px] font-semibold text-black/35">
      {selectedImages.length}/5
    </span>
  </div>

  <label className="group flex cursor-pointer items-center justify-center rounded-xl border border-dashed border-black/12 bg-white px-4 py-5 text-center">
    <div>
      <div className="mx-auto mb-1.5 grid size-8 place-items-center rounded-lg bg-black/[0.04] text-black/40">
        <Upload size={15} />
      </div>

      <p className="text-[11px] font-semibold text-black/57">
        Choose product images
      </p>

      <p className="mt-0.5 text-[9px] text-black/28">
        Select multiple files at once
      </p>
    </div>

    <input
      type="file"
      accept="image/jpeg,image/png,image/webp"
      multiple
      className="hidden"
      onChange={(event) => {
        void handleImageSelection(event.target.files);
        event.currentTarget.value = "";
      }}
    />
  </label>

  {compressingImages ? (
    <div className="mt-2 rounded-lg border border-[#d71920]/10 bg-[#d71920]/[0.04] px-3 py-2 text-[9px] font-medium text-[#d71920]">
      Optimizing images for upload...
    </div>
  ) : null}

  {selectedImages.length > 0 ? (
  <div className="mt-3 grid grid-cols-2 gap-2.5 sm:grid-cols-5">
    {selectedImages.map((image, index) => (
      <div
        key={`${image.file.name}-${index}`}
        className="overflow-hidden rounded-xl border border-black/7 bg-white"
      >
        {/* Image preview */}
        <div className="relative aspect-square bg-[#f3f3f0]">
          <Image
            src={image.preview}
            alt={image.file.name}
            fill
            sizes="(max-width: 640px) 50vw, 20vw"
            className="object-contain p-2"
            unoptimized
          />

          {/* Remove image */}
          <button
            type="button"
            onClick={() => {
              URL.revokeObjectURL(image.preview);

              setSelectedImages((current) =>
                current.filter(
                  (_, imageIndex) => imageIndex !== index,
                ),
              );
            }}
            disabled={saving || compressingImages}
            aria-label={`Remove ${image.file.name}`}
            title="Remove image"
            className="absolute right-1.5 top-1.5 grid size-7 place-items-center rounded-full bg-white/95 text-black/55 shadow-sm transition hover:bg-[#d71920] hover:text-white disabled:opacity-40"
          >
            <X size={13} />
          </button>
        </div>

        {/* Filename and file size */}
        <div className="border-t border-black/5 px-2 py-2">
          <p
            className="truncate text-[9px] font-medium text-black/65"
            title={image.file.name}
          >
            {image.file.name}
          </p>

          <p className="mt-1 text-[9px] text-black/35">
            {(image.file.size / (1024 * 1024)).toFixed(2)} MB
          </p>
        </div>
      </div>
    ))}
  </div>
) : null}
</div>
              </div>

              <div className="mt-5 flex flex-col-reverse gap-2 border-t border-black/7 pt-4 sm:flex-row sm:justify-end">
                <button
                  type="button"
                  onClick={
                    closeCreateModal
                  }
                  disabled={saving}
                  className="h-10 rounded-lg border border-black/8 px-4 text-[11px] font-semibold text-black/52 disabled:opacity-40"
                >
                  Cancel
                </button>

                <button
                  type="submit"
                  disabled={saving}
                  className="inline-flex h-10 items-center justify-center gap-2 rounded-lg bg-[#d71920] px-5 text-[11px] font-semibold text-white disabled:opacity-55"
                >
                  {saving ? (
                    <RefreshCw
                      size={13}
                      className="animate-spin"
                    />
                  ) : (
                    <Check size={13} />
                  )}

                  {saving
                    ? "Saving..."
                    : "Create product"}
                </button>
              </div>
            </form>
          </div>
        </div>
      ) : null}
    </div>
  );
}

function FormField({
  label,
  children,
}: {
  label: string;
  children: React.ReactNode;
}) {
  return (
    <label className="block">
      <span className="mb-1.5 block text-[9px] font-semibold uppercase tracking-[0.13em] text-black/36">
        {label}
      </span>

      {children}
    </label>
  );
}

function ToggleField({
  label,
  checked,
  inputRef,
  onKeyDown,
  onChange,
  type,
}: {
  label: string;
  checked: boolean;
  inputRef?: Ref<HTMLButtonElement>;
  onKeyDown?: (
    event: KeyboardEvent<HTMLButtonElement>,
  ) => void;
  onChange: (value: boolean) => void;
  type: "available" | "featured";
}) {
  const isAvailable = type === "available";

  const accent = isAvailable
    ? {
        border: "border-emerald-300/70",
        background: "bg-emerald-50",
        iconBg: "bg-emerald-100",
        iconText: "text-emerald-600",
        title: "text-emerald-700",
        switch: "bg-emerald-600",
      }
    : {
        border: "border-[#d71920]/30",
        background: "bg-[#fff5f5]",
        iconBg: "bg-[#fde4e5]",
        iconText: "text-[#d71920]",
        title: "text-[#b9141a]",
        switch: "bg-[#d71920]",
      };

  return (
    <button
      ref={inputRef}
      type="button"
      onClick={() => onChange(!checked)}
      onKeyDown={onKeyDown}
      aria-pressed={checked}
      className={`group flex min-h-[76px] w-full items-center justify-between rounded-xl border px-3.5 text-left transition-all duration-200 ${
        checked
          ? `${accent.border} ${accent.background} shadow-[0_8px_24px_rgba(0,0,0,0.05)]`
          : "border-black/8 bg-white hover:border-black/15 hover:bg-[#fafafa]"
      }`}
    >
      <div className="flex items-center gap-3">
        <span
          className={`grid size-9 shrink-0 place-items-center rounded-lg transition ${
            checked
              ? `${accent.iconBg} ${accent.iconText}`
              : "bg-black/[0.045] text-black/35"
          }`}
        >
          {isAvailable ? (
            <CircleCheck size={18} strokeWidth={2.2} />
          ) : (
            <Star
              size={17}
              strokeWidth={2.2}
              fill={checked ? "currentColor" : "none"}
            />
          )}
        </span>

        <div>
          <p
            className={`text-[11px] font-bold transition ${
              checked
                ? accent.title
                : "text-black/62"
            }`}
          >
            {label}
          </p>

          <p
            className={`mt-0.5 text-[9px] transition ${
              checked
                ? "text-black/42"
                : "text-black/30"
            }`}
          >
            {isAvailable
              ? checked
                ? "Customers can purchase this product."
                : "Product is hidden from availability."
              : checked
                ? "Product will be highlighted as featured."
                : "Product will appear as a normal item."}
          </p>
        </div>
      </div>

      <div className="flex items-center gap-2.5">
        <span

function resolveImageUrl(value: string | null | undefined) {
  if (!value) return "";

  let url = value.trim().replace(/\\/g, "/");

  if (!url) return "";

  if (
    /^https?:\/\//i.test(url) ||
    url.startsWith("data:") ||
    url.startsWith("blob:")
  ) {
    return url;
  }

  url = url.replace(/^\.\//, "");
  url = url.replace(/^public\//i, "");
  url = url.replace(/^\/public\//i, "/");

  const publicIndex = url.toLowerCase().indexOf("/public/");

  if (publicIndex >= 0) {
    url = url.slice(publicIndex + "/public/".length);
  }

  if (!url.startsWith("/")) {
    url = `/${url}`;
  }

  return url;
}

async function readJsonResponse(
  response: Response,
): Promise<Record<string, unknown>> {
  const text = await response.text();

  if (!text.trim()) {
    return {};
  }

  try {
    const parsed: unknown = JSON.parse(text);

    if (parsed && typeof parsed === "object") {
      return parsed as Record<string, unknown>;
    }

    return {};
  } catch {
    throw new Error(
      `Server returned an unexpected response (${response.status} ${response.statusText}).`,
    );
  }
}

function getResponseMessage(
  data: Record<string, unknown>,
  fallback: string,
): string {
  const message = data.message ?? data.error;

  return typeof message === "string"
    ? message
    : fallback;
}

function formatPrice(value: string | number | null) {
  if (value === null || value === "") {
    return "Contact for Price";
  }

  const amount = Number(value);

  if (!Number.isFinite(amount)) {
    return "Contact for Price";
  }

  return `LKR ${amount.toLocaleString("en-LK", {
    minimumFractionDigits: 2,
    maximumFractionDigits: 2,
  })}`;
}

export const runtime = "nodejs";

const ALLOWED_PAGE_SIZES = [25, 50, 100] as const;

type ProductStatus = "DRAFT" | "PUBLISHED" | "ARCHIVED";

function parsePositiveInteger(
  value: string | null,
  fallback: number,
) {
  const parsed = Number(value);

  return Number.isSafeInteger(parsed) && parsed > 0
    ? parsed
    : fallback;
}

const allowedStatuses = [
  "DRAFT",
  "PUBLISHED",
  "ARCHIVED",
] as const;

function makeSlug(value: string) {
  return value
    .toLowerCase()
    .trim()
    .normalize("NFKD")
    .replace(/[\u0300-\u036f]/g, "")
    .replace(/[^a-z0-9]+/g, "-")
    .replace(/^-+|-+$/g, "")
    .slice(0, 200);
}

async function getAuthorizedUser() {
  const session = await getSession();

  if (!session) {
    return {
      error: "Unauthorized",
      status: 401 as const,
    };
  }

  const user = await prisma.store_user.findFirst({
    where: {
      id: session.userId,
      storeId: session.storeId,
      isActive: true,
    },
    include: {
      store: {
        select: {
          isActive: true,
        },
      },
    },
  });

  if (!user || !user.store.isActive) {
    return {
      error: "Account or store is inactive",
      status: 403 as const,
    };
  }

  if (user.role !== "OWNER" && user.role !== "ADMIN") {
    return {
      error: "You do not have permission",
      status: 403 as const,
    };
  }

  return {
    user,
    session,
  };
}

export async function GET(request: Request) {
  try {
    const auth = await getAuthorizedUser();

    if ("error" in auth) {
      return NextResponse.json(
        { error: auth.error },
        { status: auth.status },
      );
    }

    const searchParams = new URL(request.url).searchParams;

    const requestedPage = parsePositiveInteger(
      searchParams.get("page"),
      1,
    );

    const requestedPageSize = parsePositiveInteger(
      searchParams.get("pageSize"),
      25,
    );

    const pageSize = ALLOWED_PAGE_SIZES.includes(
      requestedPageSize as (typeof ALLOWED_PAGE_SIZES)[number],
    )
      ? requestedPageSize
      : 25;

    const search = searchParams.get("search")?.trim() ?? "";

    const rawStatus = searchParams.get("status")?.trim() ?? "";

    const status: ProductStatus | "" =
      rawStatus === "DRAFT" ||
      rawStatus === "PUBLISHED" ||
      rawStatus === "ARCHIVED"
        ? rawStatus
        : "";

    const availability =
      searchParams.get("availability")?.trim() ?? "";

    const featured =
      searchParams.get("featured")?.trim() ?? "";

    const where = {
      storeId: auth.session.storeId,

      ...(search
        ? {
            OR: [
              {
                name: {
                  contains: search,
                },
              },
              {
                slug: {
                  contains: search,
                },
              },
              {
                sku: {
                  contains: search,
                },
              },
            ],
          }
        : {}),

      ...(status
        ? {
            status,
          }
        : {}),

      ...(availability === "available"
        ? {
            isAvailable: true,
          }
        : availability === "unavailable"
          ? {
              isAvailable: false,
            }
          : {}),

      ...(featured === "featured"
        ? {
            isFeatured: true,
          }
        : featured === "not-featured"
          ? {
              isFeatured: false,
            }
          : {}),
    };

    const [
      total,
      totalOverall,
      published,

function cleanString(value: unknown): string {
  return String(value ?? "").trim();
}

function nullableString(value: unknown): string | null {
  const result = cleanString(value);
  return result || null;
}

function parsePositiveInt(value: unknown): number | null {
  if (value === null || value === undefined || value === "") {
    return null;
  }

  const parsed = Number(value);

  if (!Number.isInteger(parsed) || parsed <= 0) {
    return null;
  }

  return parsed;
}

function parseMoney(value: unknown): number | null {
  if (value === null || value === undefined || value === "") {
    return null;
  }

  const parsed = Number(value);

  if (!Number.isFinite(parsed) || parsed < 0) {
    return null;
  }

  return parsed;
}

function createSlug(value: string): string {
  const base = value
    .trim()
    .toLowerCase()
    .replace(/[^a-z0-9]+/g, "-")
    .replace(/^-+|-+$/g, "");

  return base || `product-${Date.now()}`;
}

async function createUniqueSlug(
  storeId: number,
  name: string,
): Promise<string> {
  const base = createSlug(name);

  let slug = base;
  let counter = 2;

  while (
    await prisma.product.findFirst({
      where: {
        storeId,
        slug,
      },
      select: {
        id: true,
      },
    })
  ) {
    slug = `${base}-${counter}`;
    counter += 1;
  }

  return slug;
}

    ) {
      isFeatured =
        body.isFeatured;
    }

    /*
    |--------------------------------------------------------------------------
    | CATEGORY
    |
    | We validate the category before updating the product.
    |--------------------------------------------------------------------------
    */

    let categoryUpdate:
      | {
          connect: {
            id: number;
          };
        }
      | {
          disconnect: true;
        }
      | undefined;

    if (
      body.categoryId !==
      undefined
    ) {
      if (
        body.categoryId ===
          null ||
        body.categoryId ===
          ""
      ) {
        categoryUpdate = {
          disconnect: true,
        };
      } else {
        const requestedCategoryId =
          Number(
            body.categoryId,
          );

        if (
          !Number.isInteger(
            requestedCategoryId,
          ) ||
          requestedCategoryId <= 0
        ) {
          return NextResponse.json(
            {
              success: false,
              message:
                "Invalid category ID.",
            },
            {
              status: 400,
            },
          );
        }

        const category =
          await prisma.category.findUnique(
            {
              where: {
                id: requestedCategoryId,
              },

              select: {
                id: true,
              },
            },
          );

        if (!category) {
          return NextResponse.json(
            {
              success: false,
              message:
                "Selected category was not found.",
            },
            {
              status: 400,
            },
          );
        }

        categoryUpdate = {
          connect: {
            id:
              requestedCategoryId,
          },
        };
      }
    }

    /*
    |--------------------------------------------------------------------------
    | BUILD UPDATE DATA
    |--------------------------------------------------------------------------
    */

    const updateData = {
      name,
      sku,
      price,
      compareAtPrice,
      shortDescription,
      description,

      ...(status !==
      undefined
        ? {
            status,
          }
        : {}),

      ...(isAvailable !==
      undefined
        ? {
            isAvailable,
          }
        : {}),

      ...(isFeatured !==
      undefined
        ? {
            isFeatured,
          }
        : {}),

      ...(categoryUpdate !==
      undefined
        ? {
            category:
              categoryUpdate,
          }
        : {}),
    };

    /*
    |--------------------------------------------------------------------------
    | UPDATE PRODUCT
    |--------------------------------------------------------------------------
    */

    await prisma.product.update({
      where: {
        id: productId,
      },

      data: updateData,
    });

    /*
    |--------------------------------------------------------------------------
    | FETCH UPDATED PRODUCT WITH CATEGORY + IMAGES
    |--------------------------------------------------------------------------
    */

    const updatedProduct =
      await getProductWithRelations(
        productId,
      );

    if (!updatedProduct) {
      return NextResponse.json(
        {
          success: false,
          message:
            "Product was updated but could not be loaded afterward.",
        },
        {
          status: 500,
        },
      );
    }

    /*
    |--------------------------------------------------------------------------
    | RESPONSE
    |--------------------------------------------------------------------------
    */

    return NextResponse.json({
      success: true,
      message:
        "Product updated successfully.",
      product:
        mapProduct(
          updatedProduct,
        ),
    });
  } catch (error) {
    console.error(
      "Update admin product error:",
      error,
    );

    return NextResponse.json(
      {
        success: false,
        message:
          "Failed to update product.",
      },
      {
        status: 500,
      },
    );
  }
}
