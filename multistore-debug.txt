import { createHash, randomBytes } from "node:crypto";
import { cookies } from "next/headers";

import { prisma } from "@/lib/prisma";

const STORE_SESSION_COOKIE = "session_user";
const SUPER_ADMIN_SESSION_COOKIE = "super_admin_session";
const SESSION_TTL_MS = 8 * 60 * 60 * 1000;

export type StoreAdminSession = {
  userId: number;
  storeId: number;
  storeSlug: string;
  storeName: string;
  name: string;
  username: string;
  email: string;
  role: "OWNER" | "ADMIN" | "STAFF";
  actorType: "STORE_USER" | "SUPER_ADMIN";
  superAdminId: number | null;
  isImpersonating: boolean;
};

export type SuperAdminSession = {
  id: number;
  name: string;
  username: string;
  email: string;
};

function hashToken(token: string) {
  return createHash("sha256")
    .update(token)
    .digest("hex");
}

function cookieOptions(maxAge: number) {
  return {
    httpOnly: true,
    secure: process.env.NODE_ENV === "production",
    sameSite: "lax" as const,
    path: "/",
    maxAge,
  };
}

export async function createStoreAdminSession(userId: number) {
  const token = randomBytes(32).toString("hex");
  const tokenHash = hashToken(token);
  const expiresAt = new Date(Date.now() + SESSION_TTL_MS);

  await prisma.app_session.create({
    data: {
      tokenHash,
      storeUserId: userId,
      expiresAt,
    },
  });

  const cookieStore = await cookies();
  cookieStore.set(
    STORE_SESSION_COOKIE,
    token,
    cookieOptions(Math.floor(SESSION_TTL_MS / 1000)),
  );
}

export async function createSuperAdminSession(superAdminId: number) {
  const token = randomBytes(32).toString("hex");
  const tokenHash = hashToken(token);
  const expiresAt = new Date(Date.now() + SESSION_TTL_MS);

  await prisma.app_session.create({
    data: {
      tokenHash,
      superAdminId,
      expiresAt,
    },
  });

  const cookieStore = await cookies();
  cookieStore.set(
    SUPER_ADMIN_SESSION_COOKIE,
    token,
    cookieOptions(Math.floor(SESSION_TTL_MS / 1000)),
  );
}

export async function getSession(): Promise<StoreAdminSession | null> {
  const cookieStore = await cookies();
  const token = cookieStore.get(STORE_SESSION_COOKIE)?.value;

  if (!token) {
    return null;
  }

  const session = await prisma.app_session.findUnique({
    where: {
      tokenHash: hashToken(token),
    },
    include: {
      storeUser: {
        include: {
          store: {
            select: {
              id: true,
              slug: true,
              name: true,
              isActive: true,
            },
          },
        },
      },
      superAdmin: {
        select: { id: true },
      },
    },
  });

  if (
    !session ||
    !session.storeUser ||
    session.expiresAt <= new Date() ||
    !session.storeUser.isActive ||
    !session.storeUser.store.isActive
  ) {
    return null;
  }

  return {
    userId: session.storeUser.id,
    storeId: session.storeUser.store.id,
    storeSlug: session.storeUser.store.slug,
    storeName: session.storeUser.store.name,
    name: session.storeUser.name,
    username: session.storeUser.username,
    email: session.storeUser.email,
    role: session.storeUser.role,
    actorType: session.superAdminId ? "SUPER_ADMIN" : "STORE_USER",
    superAdminId: session.superAdminId ?? null,
    isImpersonating: session.superAdminId !== null,
  };
}

export async function createSuperAdminStoreOverride(superAdminId: number, storeId: number) {
  const ownerOrAdmin = await prisma.store_user.findFirst({
    where: {
      storeId,
      isActive: true,
      role: { in: ["OWNER", "ADMIN"] },
      store: { isActive: true },
    },
    orderBy: [{ role: "asc" }, { id: "asc" }],
    select: { id: true },
  });

  if (!ownerOrAdmin) {
    throw new Error("No active store administrator exists for this store.");
  }

  const token = randomBytes(32).toString("hex");
  const tokenHash = hashToken(token);
  const expiresAt = new Date(Date.now() + 60 * 60 * 1000);

  await prisma.app_session.create({
    data: {
      tokenHash,
      storeUserId: ownerOrAdmin.id,
      superAdminId,
      expiresAt,
    },
  });

  const cookieStore = await cookies();
  cookieStore.set(
    STORE_SESSION_COOKIE,
    token,
    cookieOptions(60 * 60),
  );
}

export async function getSuperAdminSession(): Promise<SuperAdminSession | null> {
  const cookieStore = await cookies();
  const token = cookieStore.get(SUPER_ADMIN_SESSION_COOKIE)?.value;

  if (!token) {
    return null;
  }

  const session = await prisma.app_session.findUnique({
    where: {
      tokenHash: hashToken(token),
    },
    include: {
      superAdmin: { select: { id: true, name: true, username: true, email: true } },
    },
  });

  if (
    !session ||
    !session.superAdmin ||
    session.expiresAt <= new Date() ||
    !session.superAdmin.isActive
  ) {
    return null;
  }

  return {
    id: session.superAdmin.id,
    name: session.superAdmin.name,
    username: session.superAdmin.username,
    email: session.superAdmin.email,
  };
}

export async function clearAuthSessions() {
  const cookieStore = await cookies();
  const storeToken = cookieStore.get(STORE_SESSION_COOKIE)?.value;
  const superToken = cookieStore.get(SUPER_ADMIN_SESSION_COOKIE)?.value;

  await Promise.all([
    storeToken
      ? prisma.app_session.deleteMany({
          where: {
            tokenHash: hashToken(storeToken),
          },
        })
      : Promise.resolve(),
    superToken
      ? prisma.app_session.deleteMany({
          where: {
            tokenHash: hashToken(superToken),
          },
        })
      : Promise.resolve(),
  ]);

  cookieStore.delete(STORE_SESSION_COOKIE);
  cookieStore.delete(SUPER_ADMIN_SESSION_COOKIE);
}

export async function exitSuperAdminStoreOverride() {
  const cookieStore = await cookies();
  const token = cookieStore.get(STORE_SESSION_COOKIE)?.value;

  if (token) {
    await prisma.app_session.deleteMany({
      where: { tokenHash: hashToken(token) },
    });
  }

  cookieStore.delete(STORE_SESSION_COOKIE);
}

export type AuditActorType = "STORE_USER" | "SUPER_ADMIN" | "SYSTEM";

export type AuditLogInput = {
  storeId?: number | null;
  storeUserId?: number | null;
  superAdminId?: number | null;
  actorType: AuditActorType;
  module: string;
  action: string;
  entity?: string | null;
  entityId?: number | string | null;
  description: string;
  metadata?: unknown;
  ipAddress?: string | null;
  userAgent?: string | null;

export async function writeAuditLog(input: AuditLogInput) {
  try {
    await prisma.audit_log.create({
      data: {
        storeId: input.storeId ?? null,
        storeUserId: input.storeUserId ?? null,
        superAdminId: input.superAdminId ?? null,
        actorType: input.actorType,
        module: input.module,
        action: input.action,
        entity: input.entity ?? null,
        entityId:
          input.entityId === undefined || input.entityId === null
            ? null
            : String(input.entityId),
        description: input.description,
        metadata: input.metadata as never,
        ipAddress: input.ipAddress ?? null,
        userAgent: input.userAgent ?? null,
      },
    });
  } catch (error) {
    console.error("Audit log write failed:", error);
  }
}

export function getRequestAuditContext(request: Request) {
  return {
    ipAddress:
      request.headers.get("x-forwarded-for")?.split(",")[0]?.trim() ??
      request.headers.get("x-real-ip") ??
      null,
    userAgent: request.headers.get("user-agent") ?? null,
  };
}

import { NextResponse } from "next/server";

import { prisma } from "@/lib/prisma";
import { getSession } from "@/lib/auth";
import { getRequestAuditContext, writeAuditLog } from "@/lib/audit";

export const runtime = "nodejs";

async function getAuthorizedStoreAdmin() {
  const session = await getSession();

  if (!session) {
    return { error: "Unauthorized", status: 401 as const };
  }

  if (session.role !== "OWNER" && session.role !== "ADMIN") {
    return { error: "You do not have permission", status: 403 as const };
  }

  const store = await prisma.store.findFirst({
    where: {
      id: session.storeId,
      isActive: true,
    },
    select: {
      id: true,
      name: true,
      slug: true,
      currency: true,
      template: true,
    },
  });

  if (!store) {
    return { error: "Store is inactive or not found", status: 403 as const };
  }

  return { session, store };
}

async function ensureConfiguration(storeId: number, storeName: string) {
  const [profile, settings, theme, store] = await Promise.all([
    prisma.store_profile.upsert({
      where: { storeId },
      create: { storeId, displayName: storeName },
      update: {},
    }),
    prisma.store_settings.upsert({
      where: { storeId },
      create: { storeId },
      update: {},
    }),
    prisma.store_theme.upsert({
      where: { storeId },
      create: { storeId },
      update: {},
    }),
    prisma.store.findUnique({
      where: { id: storeId },
      select: { template: true },
    }),
  ]);

  return { profile, settings, theme, template: store?.template ?? "MODERN" };
}

export async function GET() {
  try {
    const auth = await getAuthorizedStoreAdmin();

    if ("error" in auth) {
      return NextResponse.json({ error: auth.error }, { status: auth.status });
    }

    const config = await ensureConfiguration(auth.store.id, auth.store.name);
    const heroSlides = await prisma.hero_slide.findMany({
      where: { storeId: auth.store.id },
      orderBy: { sortOrder: "asc" },
    });

    return NextResponse.json({
      success: true,
      store: auth.store,
      profile: config.profile,
      settings: config.settings,
      theme: config.theme,
      template: config.template,
      heroSlides,
    });
  } catch (error) {
    console.error("GET store profile error:", error);
    return NextResponse.json(
      { error: "Failed to load store profile" },
      { status: 500 },
    );
  }
}

export async function PATCH(request: Request) {
  try {
    const auth = await getAuthorizedStoreAdmin();

    if ("error" in auth) {
      return NextResponse.json({ error: auth.error }, { status: auth.status });
    }

    const body: unknown = await request.json();

    if (!body || typeof body !== "object") {
      return NextResponse.json(
        { error: "Invalid request body" },
        { status: 400 },
      );
    }

    const data = body as Record<string, unknown>;
    const profile = (data.profile && typeof data.profile === "object"
      ? data.profile
      : {}) as Record<string, unknown>;
    const settings = (data.settings && typeof data.settings === "object"
      ? data.settings
      : {}) as Record<string, unknown>;
    const theme = (data.theme && typeof data.theme === "object"
      ? data.theme
      : {}) as Record<string, unknown>;

    const template = String(data.template ?? auth.store.template);
    const allowedTemplates = ["MODERN", "CLASSIC", "MINIMAL"];

      return NextResponse.json(
        { error: "Invalid storefront template" },
        { status: 400 },

      !Number.isInteger(heroOfferPercent) ||
      heroOfferPercent < 0 ||
      heroOfferPercent > 100 ||
      !Number.isFinite(freeDeliveryThreshold) ||
      freeDeliveryThreshold < 0 ||
      !Number.isFinite(deliveryCharge) ||
      deliveryCharge < 0 ||
      !Number.isFinite(minimumOrder) ||
      minimumOrder < 0
    ) {
      return NextResponse.json(
        { error: "Invalid storefront numeric settings" },
        { status: 400 },
      );

        where: { storeId: auth.store.id },
        create: {
          storeId: auth.store.id,
          displayName: String(profile.displayName ?? updatedStore.name).trim().slice(0, 150),
        },
        update: {},
      });

      const profileData = {
        displayName: String(profile.displayName ?? updatedProfile.displayName).trim().slice(0, 150),
        tagline: String(profile.tagline ?? "").trim() || null,
        description: String(profile.description ?? "").trim() || null,
        phone: String(profile.phone ?? "").trim() || null,
        whatsapp: String(profile.whatsapp ?? "").trim() || null,
        email: String(profile.email ?? "").trim().toLowerCase() || null,
        supportEmail: String(profile.supportEmail ?? "").trim().toLowerCase() || null,
        addressLine1: String(profile.addressLine1 ?? "").trim() || null,
        addressLine2: String(profile.addressLine2 ?? "").trim() || null,
        city: String(profile.city ?? "").trim() || null,
        district: String(profile.district ?? "").trim() || null,
        province: String(profile.province ?? "").trim() || null,
        postalCode: String(profile.postalCode ?? "").trim() || null,
        country: String(profile.country ?? "").trim() || null,
        facebookUrl: String(profile.facebookUrl ?? "").trim() || null,

import { NextResponse } from "next/server";
import bcrypt from "bcryptjs";

import { prisma } from "@/lib/prisma";
import { createStoreAdminSession } from "@/lib/auth";
import { getRequestAuditContext, writeAuditLog } from "@/lib/audit";

export const runtime = "nodejs";

export async function POST(request: Request) {
  try {
    const body: unknown = await request.json();

    const username =
      body && typeof body === "object" && "username" in body
        ? String((body as { username?: unknown }).username ?? "")
            .trim()
            .toLowerCase()
        : body && typeof body === "object" && "identifier" in body
          ? String((body as { identifier?: unknown }).identifier ?? "")
              .trim()
              .toLowerCase()
          : "";

    const password =
      body && typeof body === "object" && "password" in body
        ? String((body as { password?: unknown }).password ?? "")
        : "";

    if (!username || !password) {
      return NextResponse.json(
        {
          success: false,
          message: "Username and password are required.",
        },
        { status: 400 },
      );
    }

    const user = await prisma.store_user.findFirst({
      where: {
        OR: [
          { username },
          { email: username },
        ],
      },
      include: {
        store: {
          select: {
            id: true,
            name: true,
            slug: true,
            isActive: true,
          },
        },
      },
    });

    if (!user || !user.isActive || !user.store.isActive) {
      return NextResponse.json(
        {
          success: false,
          message: "Invalid username or password.",
        },
        { status: 401 },
      );
    }

    const passwordValid = await bcrypt.compare(
      password,
      user.passwordHash,
    );

    if (!passwordValid) {
      return NextResponse.json(
        {
          success: false,
          message: "Invalid username or password.",
        },
        { status: 401 },
      );
    }

    await createStoreAdminSession(user.id);

    const auditContext = getRequestAuditContext(request);

    await writeAuditLog({
      storeId: user.store.id,
      storeUserId: user.id,
      actorType: "STORE_USER",
      module: "AUTH",
      action: "LOGIN",
      entity: "STORE_USER",
      entityId: user.id,
      description: `${user.name} signed in to ${user.store.name}.`,
      ...auditContext,
    });

    return NextResponse.json({
      success: true,
      message: "Login successful.",
      user: {
        id: user.id,
        name: user.name,
        username: user.username,
        email: user.email,
        role: user.role,
        storeId: user.store.id,
        storeSlug: user.store.slug,
      },
    });
  } catch (error) {
    console.error("Store admin login error:", error);

    return NextResponse.json(
      {
        success: false,
        message: "Unable to process login.",
      },
      { status: 500 },
    );
  }
}
